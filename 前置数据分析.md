# 前置数据分析

### 口径

- 非同省聊天意愿：

|高|非同省有效会话数≥5 或者 非同省付费超过≥500趣币|
|---|---|
|中|非同省有效会话数≥3 或者 非同省付费超过≥100趣币|
|低|非同省有效会话数≥1 或者 非同省付费超过＞0趣币|
|无|有非同省会话，但是无有效会话或者不付费|
|未知|没有非同省会话|



- 供需程度：

重复牵线次数占比超过60%，视为供需不足，口径简单，日志问题后续优化

理论上要补充：常规召回用户数（过滤后）低于平均值或者某个阈值，也应该认为供需不足

                                            OR (link_cnt > 0
                                            AND repeat_link_cnt / link_cnt > 0.6) THEN '供需不足'
                                        WHEN COALESCE(link_cnt,0) = 0 THEN '无牵线未知'
                                        ELSE '供需暂时充足'
                                END AS 供需程度
                        FROM    taqu_algor.dws_user_distance_di
                        WHERE   dt >= '20260119'
                        AND     last_group_type RLIKE '男'
                    )
            GROUP BY account_uuid
        )
GROUP BY last_group_type,IF(中高意愿次数 > 0,1,0)
         ,IF(供需不足次数 > 0,1,0)
;

SELECT  t1.*
        ,round(cnt / all_cnt,4) AS 用户数占比
FROM    (
            SELECT  dt
                    ,非同省聊天意愿
                    ,供需程度
                    ,COUNT(1) AS cnt
                    ,1 AS grop
                    ,last_group_type
            FROM    (
                        SELECT  *
                                ,CASE   WHEN not_same_provinve_valid_conv_cnt >= 5 OR not_same_provinve_male_consume_amt_1d >= 300 THEN '高'
                                        WHEN not_same_provinve_valid_conv_cnt >= 2 OR not_same_provinve_male_consume_amt_1d >= 100 THEN '中'
                                        WHEN not_same_provinve_valid_conv_cnt > 0 OR not_same_provinve_male_consume_amt_1d > 00 THEN '低'
                                        WHEN (COALESCE(not_same_provinve_valid_conv_cnt,0) <= 0
                                            OR COALESCE(not_same_provinve_male_consume_amt_1d,0) <= 0)
                                            AND not_same_provinve_conv_cnt > 0 THEN '无'
                                        ELSE '未知'
                                END AS 非同省聊天意愿
                                ,CASE   WHEN (same_provinve_link_cnt > 0
                                            AND same_provinve_repeat_link_cnt / same_provinve_link_cnt > 0.6)
                                            OR (link_cnt > 0
                                            AND repeat_link_cnt / link_cnt > 0.6) THEN '供需不足'
                                        WHEN COALESCE(link_cnt,0) = 0 THEN '无牵线未知'
                                        ELSE '供需暂时充足'
                                END AS 供需程度
                        FROM    taqu_algor.dws_user_distance_di
                        WHERE   dt >= '20260119'
                        AND     last_group_type RLIKE '男'
                    ) 
            GROUP BY dt
                     ,非同省聊天意愿
                     ,供需程度
                     ,last_group_type
        ) t1
LEFT OUTER JOIN (
                    SELECT  dt
                            ,last_group_type
                            ,COUNT(1) AS all_cnt
                            ,1 AS grop
                    FROM    taqu_algor.dws_user_distance_di
                    WHERE   dt >= '20260119'
                    AND     last_group_type RLIKE '男'
                    GROUP BY dt
                             ,last_group_type
                ) t2
ON      t1.dt = t2.dt
AND     t1.last_group_type = t2.last_group_type
WHERE 非同省聊天意愿 = '高'
AND 供需程度 = '供需不足'
AND t1.last_group_type = '社交-5腰部男'
;
